        ) {
          detail =
            "Firebase ڕێگەی نووسین نادات. Firestore Rules پشکنە.";
        } else if (message === "AUTH_REQUIRED") {
          detail = "تکایە دووبارە login بکە.";
        } else if (message === "NAME_REQUIRED") {
          detail = "ناوی بەرهەم پڕبکەرەوە.";
        } else if (
          message.startsWith("TIMEOUT:FIRESTORE_UPDATE")
        ) {
          detail =
            "نوێکردنەوەی بەرهەم وەڵامی Firebase وەرنەگرت. زۆرجار ئەمە لە Firestore Rules یان پەیوەندی Firebase ـە.";
        } else if (
          message.startsWith("TIMEOUT:FIRESTORE_CREATE")
        ) {
          detail =
            "زیادکردنی بەرهەم وەڵامی Firebase وەرنەگرت. زۆرجار ئەمە لە Firestore Rules یان پەیوەندی Firebase ـە.";
        } else {
          detail =
            `هەڵە: ${code || "UNKNOWN"} — ${message}`;
        }
        $("msg").innerHTML = `
          <div class="error">
            پاشەکەوتکردن سەرکەوتوو نەبوو.
            <br>
            ${escapeHtml(detail)}
            <br>
            <small>
              کۆدی هەڵە:
              ${escapeHtml(code || "none")}
            </small>
          </div>
        `;
      } finally {
        button.disabled = false;
        button.textContent = "پاشەکەوتکردن";
      }
    }
    async function removeProduct(id) {
      const product = getProductById(id);
      if (!product) {
        return;
      }
      const confirmed = confirm(
        `دڵنیایت لە سڕینەوەی «${product.name}»؟`
      );
      if (!confirmed) {
        return;
      }
      try {
        await deleteDoc(doc(db, "products", id));
        showToast("سڕایەوە");
      } catch (error) {
        console.error("PRODUCT DELETE ERROR:", error);
        showToast("سڕینەوە سەرکەوتوو نەبوو");
      }
    }
    async function toggleProduct(id) {
      const product = getProductById(id);
      if (!product) {
        return;
      }
      const newActiveState =
        product.active === false;
      try {
        await updateDoc(
          doc(db, "products", id),
          {
            active: newActiveState,
            updatedAt: serverTimestamp()
          }
        );
        showToast(
          newActiveState
            ? "چالاک کرا"
            : "ناچالاک کرا"
        );
      } catch (error) {
        console.error("PRODUCT STATUS ERROR:", error);
        showToast("گۆڕینی دۆخ سەرکەوتوو نەبوو");
      }
    }
    function subscribeToProducts() {
      if (unsubscribeProducts) {
        unsubscribeProducts();
        unsubscribeProducts = null;
      }
      setSyncState(false, "چاوەڕوانی sync...");
      unsubscribeProducts = onSnapshot(
        productsRef,
        snapshot => {
          products = snapshot.docs.map(document => ({
            id: document.id,
            ...document.data()
          }));
          setSyncState(true, "هاوکاتە");
          render();
        },
        error => {
          console.error("PRODUCT SYNC ERROR:", error);
          setSyncState(false, "کێشەی sync");
        }
      );
    }
    function handleProductAction(event) {
      const button = event.target.closest("[data-action]");
      if (!button) {
        return;
      }
      const action = button.dataset.action;
      const id = button.dataset.id;
      if (action === "edit") {
        const product = getProductById(id);
        if (product) {
          openModal(product);
        }
        return;
      }
      if (action === "toggle") {
        toggleProduct(id);
        return;
      }
      if (action === "delete") {
        removeProduct(id);
      }
    }
    function handleSizeAction(event) {
      const button = event.target.closest(
        '[data-action="remove-size"]'
      );
      if (!button) {
        return;
      }
      button.closest(".size-row")?.remove();
    }

        .value
        .split(/\r?\n/)
        .map(url => url.trim())
        .filter(Boolean);
      $("preview").innerHTML = "";
      urls.forEach(addPreview);
    }

      setupEventListeners();

          showLogin();
          if (unsubscribeProducts) {
            unsubscribeProducts();
            unsubscribeProducts = null;
          }
          products = [];
          render();
          return;
        }
        showApp();
        loadBanner();
        subscribeToProducts();
      });
    }
