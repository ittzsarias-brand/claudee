          code === "permission-denied" ||

          detail = "تکایە دووبارە login بکە.";
        } else if (message === "NAME_REQUIRED") {
          detail = "ناوی بەرهەم پڕبکەرەوە.";
        } else if (
          message.startsWith("TIMEOUT:FIRESTORE_UPDATE")
        ) {
          detail =
            "نوێکردنەوەی بەرهەم وەڵامی Firebase وەرنەگرت. زۆرجار ئەمە لە Firestore Rules یان پەیوەندی Firebase ـە.";
        } else if (
          message.startsWith("TIMEOUT:FIRESTORE_CREATE")
        ) {
          detail =
            "زیادکردنی بەرهەم وەڵامی Firebase وەرنەگرت. زۆرجار ئەمە لە Firestore Rules یان پەیوەندی Firebase ـە.";

          detail =
            `هەڵە: ${code || "UNKNOWN"} — ${message}`;
        }
        $("msg").innerHTML = `
          <div class="error">
            پاشەکەوتکردن سەرکەوتوو نەبوو.

            ${escapeHtml(detail)}
            <br>
            <small>

              ${escapeHtml(code || "none")}
            </small>
          </div>
        `;
      } finally {
        button.disabled = false;
        button.textContent = "پاشەکەوتکردن";
      }
    }

        showToast("سڕایەوە");
      } catch (error) {
        console.error("PRODUCT DELETE ERROR:", error);
        showToast("سڕینەوە سەرکەوتوو نەبوو");
      }
    }
    async function toggleProduct(id) {
      const product = getProductById(id);
      if (!product) {
        return;
      }
      const newActiveState =
        product.active === false;
      try {
        await updateDoc(
          doc(db, "products", id),
          {
            active: newActiveState,
            updatedAt: serverTimestamp()
          }
        );
        showToast(
          newActiveState
            ? "چالاک کرا"
            : "ناچالاک کرا"
        );
      } catch (error) {
        console.error("PRODUCT STATUS ERROR:", error);
        showToast("گۆڕینی دۆخ سەرکەوتوو نەبوو");
      }
    }
    function subscribeToProducts() {
      if (unsubscribeProducts) {
        unsubscribeProducts();
        unsubscribeProducts = null;
      }
      setSyncState(false, "چاوەڕوانی sync...");
      unsubscribeProducts = onSnapshot(
        productsRef,
        snapshot => {
          products = snapshot.docs.map(document => ({
            id: document.id,
            ...document.data()
          }));
          setSyncState(true, "هاوکاتە");
          render();
        },
        error => {
          console.error("PRODUCT SYNC ERROR:", error);
          setSyncState(false, "کێشەی sync");
        }
      );
    }
    function handleProductAction(event) {
      const button = event.target.closest("[data-action]");
      if (!button) {
        return;
      }
      const action = button.dataset.action;
      const id = button.dataset.id;
      if (action === "edit") {
        const product = getProductById(id);
        if (product) {
          openModal(product);
        }
        return;
      }
      if (action === "toggle") {
        toggleProduct(id);
        return;
      }
      if (action === "delete") {
        removeProduct(id);
      }
    }
    function handleSizeAction(event) {
      const button = event.target.closest(
        '[data-action="remove-size"]'
      );
      if (!button) {
        return;
      }
      button.closest(".size-row")?.remove();
    }
    function updateImagePreview() {
      const urls = $("images")
        .value
        .split(/\r?\n/)
        .map(url => url.trim())
        .filter(Boolean);
      $("preview").innerHTML = "";
      urls.forEach(addPreview);
    }

        const email = $("email").value.trim();

        button.disabled = true;
        button.textContent = "چاوەڕوان بە...";
        try {

            auth,
            email,

          );
        } catch (error) {

            <div class="error">

          button.textContent = "چوونەژوورەوە";
        }
      });

          $("loginBtn").click();
        }
      });

        signOut(auth).catch(error => {
          console.error("LOGOUT ERROR:", error);
        });
      });

      $("bannerUrl").addEventListener("input", updateBannerPreview);

      $("resetBanner").addEventListener("click", resetBanner);

        render();
      });
      $("inactiveStat").addEventListener("click", () => {
        $("status").value = $("status").value === "inactive" ? "all" : "inactive";

      });
      $("search").addEventListener("input", render);

      $("status").addEventListener("change", render);

      $("form").addEventListener("submit", saveProduct);

      });

        "input",

          ? "تەنها URL ـی وێنە پشتیوانی دەکرێت؛ فایلەکان پاشەکەوت ناکرێن"
          : "URL ـی وێنە لە خوارەوە دابنێ";
        $("files").value = "";
      });
      document.addEventListener("keydown", event => {
        if (
          event.key === "Escape" &&
          !$("modal").classList.contains("hidden")
        ) {
          closeModal();
        }
      });
    }
    function initialize() {
      setupEventListeners();
      onAuthStateChanged(auth, user => {
        if (!user) {
          showLogin();
          if (unsubscribeProducts) {
            unsubscribeProducts();
            unsubscribeProducts = null;
          }
          products = [];
          render();
          return;
        }
        showApp();
        loadBanner();
        subscribeToProducts();
      });
    }
    initialize();
